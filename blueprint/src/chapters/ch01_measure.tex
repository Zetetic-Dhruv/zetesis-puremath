\chapter{Measure theory}\label{chap:measure}

\section{Analytic measurability}\label{sec:analytic-measurability}

\begin{thm}[Analytic null-measurability]
\label{thm:analytic-null-measurable}
  \lean{MeasureTheory.AnalyticSet.nullMeasurableSet}
  \leanok
Let $\alpha$ be a Polish space, $\mu$ a finite Borel measure on
$\alpha$, and $S \subseteq \alpha$ analytic. Then $S$ is
$\mu$-null-measurable \cite[Thm 29.5]{kechris1995classical},
\cite[\S7.4.11]{bogachev2007}.
\end{thm}

\begin{thm}[Inner approximation by compacts]
\label{thm:analytic-inner-approx}
  \lean{MeasureTheory.AnalyticSet.exists\_isCompact\_measureReal\_gt}
  \leanok
Under the hypotheses of Theorem~\ref{thm:analytic-null-measurable},
for every $\delta > 0$ there exists a compact $K \subseteq S$ with
$\mu(S) - \mu(K) < \delta$ (in $\R$-valued form)
\cite[Thm 17.11]{kechris1995classical}.
\end{thm}

\section{Choquet capacity}\label{sec:choquet-capacity}

\begin{defn}[Choquet capacity]
\label{def:choquet-capacity}
  \lean{MeasureTheory.IsChoquetCapacity}
  \leanok
A map $I : \mathrm{Set}(\alpha) \to \R_{\ge 0}^\infty$ is a
\emph{Choquet capacity} iff (i) it is monotone, (ii) continuous from
below along increasing sequences, and (iii) continuous from above
along antitone sequences of closed sets
\cite{choquet1955}, \cite[Defn 30.1]{kechris1995classical},
\cite[III.1]{dellacherie-meyer}.
Every finite Borel measure on a Polish space is a Choquet capacity.
\end{defn}

\begin{defn}[Compact capacity hull]
\label{def:compact-cap}
  \lean{MeasureTheory.compactCap}
  \leanok
For $\mu$ and $S$ as above,
\[
  \mathrm{compactCap}\,\mu\,S \;:=\; \sup \{ \mu(K) : K \subseteq S,\; K \text{ compact} \}.
\]
\end{defn}

\begin{thm}[Every finite Borel measure is a Choquet capacity]
\label{thm:measure-isChoquetCapacity}
  \lean{MeasureTheory.measure\_isChoquetCapacity}
  \leanok
  \uses{def:choquet-capacity}
Every finite Borel measure on a Polish space satisfies the three
axioms of Definition~\ref{def:choquet-capacity}
\cite[III.7]{dellacherie-meyer}, \cite[Thm 17.11]{kechris1995classical}.
\end{thm}

\begin{thm}[Monotonicity of \texttt{compactCap}]
\label{thm:compactCap-mono}
  \lean{MeasureTheory.compactCap\_mono}
  \leanok
  \uses{def:compact-cap}
$S \subseteq T \Rightarrow \mathrm{compactCap}\,\mu\,S \le \mathrm{compactCap}\,\mu\,T$.
\end{thm}

\begin{thm}[Measurable inner approximation]
\label{thm:measurable-compactCap-eq}
  \lean{MeasureTheory.MeasurableSet\_compactCap\_eq}
  \leanok
  \uses{def:compact-cap}
If $S$ is measurable then $\mu(S) = \mathrm{compactCap}\,\mu\,S$
\cite[Thm 17.11]{kechris1995classical}.
\end{thm}

\begin{thm}[Kechris 30.13: analytic capacitability]
\label{thm:analytic-capacitability}
  \lean{MeasureTheory.AnalyticSet.cap\_eq\_iSup\_isCompact}
  \leanok
  \uses{def:choquet-capacity, def:compact-cap}
Let $I$ be a Choquet capacity on a Polish space $\alpha$, and $S$ an
analytic subset of $\alpha$. Then
\[
  I(S) \;=\; \sup \{ I(K) : K \subseteq S,\; K \text{ compact} \}
\]
\cite[Thm 30.13]{kechris1995classical}, \cite{choquet1955},
\cite[III.29--III.33]{dellacherie-meyer}.
Together with Definition~\ref{def:compact-cap} and
Theorem~\ref{thm:measure-isChoquetCapacity}, this specializes to
$\mu(S) = \mathrm{compactCap}\,\mu\,S$.
\end{thm}

\begin{thm}[Analytic compact-hull equality]
\label{thm:analytic-compactCap-eq}
  \lean{MeasureTheory.AnalyticSet.compactCap\_eq}
  \leanok
  \uses{thm:analytic-capacitability}
For analytic $S$, $\mu(S) = \mathrm{compactCap}\,\mu\,S$. Corollary
of Theorem~\ref{thm:analytic-capacitability}
\cite[Thm 30.13]{kechris1995classical}.
\end{thm}

\section{Baire-space cylinder infrastructure}\label{sec:cylinders}

The proof of Theorem~\ref{thm:analytic-capacitability} requires a
small theory of cylinders in $\N^\N$, adapted from
\cite[\S7.A]{kechris1995classical}.

\begin{defn}[Cylinder]
\label{def:cyl}
  \lean{Cyl}
  \leanok
For $N : \N \to \N$ and $n \in \N$,
\[
  \mathrm{Cyl}\,N\,n \;:=\; \{ g : \N \to \N \mid \forall i \le n,\; g\,i \le N\,i \}
\]
\cite[\S7.A]{kechris1995classical}.
\end{defn}

\begin{defn}[Bounded box]
\label{def:bnd}
  \lean{Bnd}
  \leanok
  \uses{def:cyl}
$\mathrm{Bnd}\,N \;:=\; \bigcap_n \mathrm{Cyl}\,N\,n \;=\; \prod_i [0, N_i]$
\cite[\S7.A]{kechris1995classical}.
\end{defn}

\begin{defn}[Truncation map]
\label{def:truncate}
  \lean{truncate}
  \leanok
$\mathrm{truncate}\,N\,g\,i \;:=\; \min(g\,i,\; N\,i)$. Sends any
sequence into $\mathrm{Bnd}\,N$ and agrees with $g$ on cylinders.
\end{defn}

\begin{lemma}[$\mathrm{Bnd}\,N$ is compact]
\label{lem:isCompact-bnd}
  \lean{isCompact\_bnd}
  \leanok
  \uses{def:bnd}
$\mathrm{Bnd}\,N$ is compact (Tychonoff applied to the finite
intervals $[0, N_i]$).
\end{lemma}

\begin{lemma}[$\mathrm{Bnd} \subseteq \mathrm{Cyl}$]
\label{lem:bnd-subset-cyl}
  \lean{bnd\_subset\_cyl}
  \leanok
  \uses{def:bnd, def:cyl}
$\mathrm{Bnd}\,N \subseteq \mathrm{Cyl}\,N\,n$ for every $n$.
\end{lemma}

\begin{lemma}[Cylinder successor splits along the next coordinate]
\label{lem:cyl-succ-eq}
  \lean{cyl\_succ\_eq}
  \leanok
  \uses{def:cyl}
$\mathrm{Cyl}\,N\,(n+1) = \mathrm{Cyl}\,N\,n \cap \{g : g\,(n+1) \le N\,(n+1)\}$.
\end{lemma}

\begin{lemma}[Intersection with a coordinate update]
\label{lem:cyl-inter-update}
  \lean{cyl\_inter\_eq\_cyl\_update}
  \leanok
  \uses{def:cyl}
Intersecting a cylinder with a coordinate-level constraint rewrites
as a cylinder with an updated bound at that coordinate.
\end{lemma}

\begin{lemma}[Monotone split]
\label{lem:monotone-cyl-split}
  \lean{monotone\_cyl\_split}
  \leanok
  \uses{def:cyl}
The map $k \mapsto \mathrm{Cyl}\,N\,n \cap \{g : g\,(n+1) \le k\}$
is monotone in $k$.
\end{lemma}

\begin{lemma}[Truncate lands in $\mathrm{Bnd}$]
\label{lem:truncate-mem-bnd}
  \lean{truncate\_mem\_bnd}
  \leanok
  \uses{def:truncate, def:bnd}
$\mathrm{truncate}\,N\,g \in \mathrm{Bnd}\,N$ for every $g$.
\end{lemma}

\begin{lemma}[Truncate agrees on cylinders]
\label{lem:truncate-agree-on-cyl}
  \lean{truncate\_agree\_on\_cyl}
  \leanok
  \uses{def:truncate, def:cyl}
If $g \in \mathrm{Cyl}\,N\,n$ then $\mathrm{truncate}\,N\,g\,i = g\,i$
for all $i \le n$.
\end{lemma}

\begin{lemma}[Cylinder extensionality]
\label{lem:cyl-ext}
  \lean{cyl\_ext}
  \leanok
  \uses{def:cyl}
Two elements of $\mathrm{Cyl}\,N\,n$ agreeing on the first $n+1$
coordinates are equal on $\mathrm{Cyl}$.
\end{lemma}

\begin{lemma}[Intersection of cylinder-closure images equals compact image]
\label{lem:iInter-closure-image-cyl-eq}
  \lean{iInter\_closure\_image\_cyl\_eq}
  \leanok
  \uses{def:cyl, def:bnd}
Under a continuous parametrization $f : \N^\N \to \alpha$ into a
Polish space,
$\bigcap_n \overline{f(\mathrm{Cyl}\,N\,n)} = f(\mathrm{Bnd}\,N)$.
\end{lemma}

\section{Total variation distance}\label{sec:tv}

\begin{defn}[Real-valued total variation]
\label{def:tv-dist-real}
  \lean{MeasureTheory.tvDistReal}
  \leanok
For finite Borel measures $P, Q$ on $\alpha$,
\[
  \TV(P, Q) \;:=\; \sup_A \bigl| P(A).\mathrm{toReal} - Q(A).\mathrm{toReal} \bigr|
\]
where the supremum is over measurable $A \subseteq \alpha$. This is
the \emph{half-sup} / metric convention, chosen so that Pinsker
reads as $\TV(P,Q) \le \sqrt{\KL(P\|Q)/2}$
\cite[\S3]{bogachev2007}, \cite[Ch.~1]{villani2009}.
\end{defn}

\begin{lemma}[Set nonempty]
\label{lem:tvDistReal-set-nonempty}
  \lean{MeasureTheory.tvDistReal\_set\_nonempty}
  \leanok
  \uses{def:tv-dist-real}
The set $\{ |P(A).\mathrm{toReal} - Q(A).\mathrm{toReal}| : A \text{ measurable} \}$
is nonempty (witnessed by $A = \emptyset$).
\end{lemma}

\begin{lemma}[Set bounded above]
\label{lem:tvDistReal-set-bddAbove}
  \lean{MeasureTheory.tvDistReal\_set\_bddAbove}
  \leanok
  \uses{def:tv-dist-real}
The same set is bounded above by $P(\alpha).\mathrm{toReal} + Q(\alpha).\mathrm{toReal}$.
\end{lemma}

\begin{remark}
The repository also contains a second total-variation object,
$\mathrm{FintypePMF.tvDistance}\,p\,q = \sum_a |p\,a - q\,a|$ (the
$L^1$ convention, see Chapter~\ref{chap:probability}); the two
conventions differ by a factor of $2$ on discrete supports.
Downstream consumers should not compose them without inserting the
factor-of-$\tfrac{1}{2}$ correction.
\end{remark}
